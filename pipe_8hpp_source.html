12// `pipe` applies a unary callable to a value, spelled infix.
13//
14// Most range and sender adaptors take their subject as the first argument --
15// `views::filter(r, pred)`, `then(sndr, fn)` -- so the backtick spells them
16// directly: `r `views::filter` pred`. The ones that take no argument beyond
17// the subject have no such spelling, because a backtick operator is exactly
18// binary. `views::join` and `sync_wait` are already the whole callable, so
19// there is no second operand to write.
20//
21// `pipe` supplies one. `x `pipe` f` is `f(x)`, which keeps a niladic stage in
22// the same left-to-right chain as its neighbours instead of forcing the chain
23// to be turned inside out into a call.
24inline constexpr auto pipe =
25 []<typename X, typename F>(X &&x, F &&f) -> decltype(auto) {
26 return std::invoke(std::forward<F>(f), std::forward<X>(x));
27};
28// 17cc4d61-f8c7-460f-880e-6f91fcf875b5 end

30} // namespace smd::infix
